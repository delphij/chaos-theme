{{- $lines := slice -}}
{{- $sec := site.Params.security -}}
{{- if or $sec.enable $sec.contact site.Params.author.email -}}
  {{- /* Contact (RFC 9116 Section 2.5.3: MUST be present) */ -}}
  {{- /* A field that may repeat takes a string or a list; appending either to
         an empty slice gives a list. */ -}}
  {{- if $sec.contact -}}
    {{- range slice | append $sec.contact -}}
      {{- $lines = $lines | append (printf "Contact: %s" .) -}}
    {{- end -}}
  {{- else if site.Params.author.email -}}
    {{- $lines = $lines | append (printf "Contact: mailto:%s" site.Params.author.email) -}}
  {{- end -}}

  {{- /* Expires (RFC 9116 Section 2.5.5: MUST be present, RFC 3339) */ -}}
  {{- $expires := $sec.expires -}}
  {{- if not $expires -}}
    {{- $expires = now.UTC.AddDate 1 0 0 | time.Format "2006-01-02T15:04:05.000Z" -}}
  {{- end -}}
  {{- $lines = $lines | append (printf "Expires: %s" $expires) -}}

  {{- /* Preferred-Languages (RFC 9116 Section 2.5.8) */ -}}
  {{- if $sec.languages -}}
    {{- $lines = $lines | append (printf "Preferred-Languages: %s" (delimit (slice | append $sec.languages) ", ")) -}}
  {{- else -}}
    {{- $lines = $lines | append (printf "Preferred-Languages: %s" site.Language.Locale) -}}
  {{- end -}}

  {{- /* Canonical (RFC 9116 Section 2.5.2) */ -}}
  {{- $canonical := $sec.canonical | default (printf "/.well-known/security.txt" | absURL) -}}
  {{- $lines = $lines | append (printf "Canonical: %s" $canonical) -}}

  {{- /* Encryption (RFC 9116 Section 2.5.4) */ -}}
  {{- range slice | append ($sec.encryption | default slice) -}}
    {{- $lines = $lines | append (printf "Encryption: %s" .) -}}
  {{- end -}}

  {{- /* Policy (RFC 9116 Section 2.5.7) */ -}}
  {{- with $sec.policy -}}
    {{- $lines = $lines | append (printf "Policy: %s" .) -}}
  {{- end -}}

  {{- /* Acknowledgments (RFC 9116 Section 2.5.1) */ -}}
  {{- range slice | append ($sec.acknowledgments | default slice) -}}
    {{- $lines = $lines | append (printf "Acknowledgments: %s" .) -}}
  {{- end -}}

  {{- /* Hiring (RFC 9116 Section 2.5.6) */ -}}
  {{- with $sec.hiring -}}
    {{- $lines = $lines | append (printf "Hiring: %s" .) -}}
  {{- end -}}
{{- end -}}
{{ delimit $lines "\n" }}
