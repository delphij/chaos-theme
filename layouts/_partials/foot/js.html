{{- /* Emits the site's scripts through js.Build: minified and fingerprinted with an integrity hash in production, unminified with an external source map in development. A loop rather than one block per script, so the build options cannot drift apart from each other. @example: {{ partialCached "foot/js.html" . }} */ -}} {{- $scripts := slice "js/main.js" -}} {{- if site.Params.search.enable -}} {{- $scripts = $scripts | append "js/search.js" -}} {{- end -}} {{- range $scripts }} {{- $opts := dict "minify" (not hugo.IsDevelopment) "sourceMap" (cond hugo.IsDevelopment "external" "") "targetPath" . }} {{- with resources.Get . }} {{- with . | js.Build $opts }} {{- if hugo.IsDevelopment }} {{- else }} {{- with . | fingerprint }} {{- end }} {{- end }} {{- end }} {{- end }} {{- end }}
