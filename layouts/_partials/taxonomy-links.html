{{- /* Renders a post's links for one taxonomy: its categories, comma-separated, or its tags, each prefixed with '#'. A tag that repeats one of the post's categories is left out. @param {page} page The post. @param {string} taxonomy "categories" or "tags". @example: {{ partial "taxonomy-links.html" (dict "page" . "taxonomy" "tags") }} */ -}} {{- $page := .page -}} {{- $taxonomy := .taxonomy -}} {{- $isTags := eq $taxonomy "tags" -}} {{- if ne $page.Type "page" -}} {{- /* GetTerms keeps front matter order and gives each term's own page, so its URL and canonical title come without a site.GetPage per name. */ -}} {{- $terms := $page.GetTerms $taxonomy -}} {{- $cats := slice -}} {{- if $isTags -}} {{- range $page.GetTerms "categories" -}} {{- $cats = $cats | append .LinkTitle -}} {{- end -}} {{- end -}} {{- $i := 0 -}} {{- range $terms -}} {{- $name := .LinkTitle -}} {{- if $isTags -}} {{- if not (in $cats $name) -}} #{{ $name }} {{- end -}} {{- else -}} {{- if $i -}}, {{- end -}} {{ $name }} {{- end -}} {{- $i = add $i 1 -}} {{- end -}} {{- end -}}
