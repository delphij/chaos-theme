{{- /* Renders taxonomy links (categories and tags) for a post. Displays clickable category and tag links with appropriate separators. Categories appear first, followed by tags prefixed with '#'. Duplicate tags that match category names are automatically filtered out. Supports optional tag limit for summary cards. @context {page|map} page The page context or dict with "page" and optional "limit". @example: {{ partial "taxonomy-links.html" . }} @example: {{ partial "taxonomy-links.html" (dict "page" . "limit" 3) }} */}} {{- $page := . -}} {{- $limit := 0 -}} {{- $categoriesOnly := false -}} {{- $tagsOnly := false -}} {{- if reflect.IsMap . -}} {{- $page = .page -}} {{- $limit = .limit | default 0 -}} {{- $categoriesOnly = .categoriesOnly | default false -}} {{- $tagsOnly = .tagsOnly | default false -}} {{- end -}} {{- if and $page (ne $page.Type "page") -}} {{- $categories := $page.Params.categories | default slice -}} {{- if not $tagsOnly -}} {{- with $page.Params.categories -}} {{- range $i, $cat := . -}} {{- if gt $i 0 -}}, {{- end -}} {{ $cat }} {{- end -}} {{- end -}} {{- end -}} {{- if not $categoriesOnly -}} {{- with $page.Params.tags -}} {{- $count := 0 -}} {{- range . -}} {{- if not (in $categories .) -}} {{- if or (le $limit 0) (lt $count $limit) -}} #{{ . }} {{- $count = add $count 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
