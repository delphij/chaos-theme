{{- /* Renders a post's links for one taxonomy: its categories, comma-separated, or its tags, each prefixed with '#'. A tag that repeats one of the post's categories is left out. @param {page} page The post. @param {string} taxonomy "categories" or "tags". @example: {{ partial "taxonomy-links.html" (dict "page" . "taxonomy" "tags") }} */ -}} {{- $page := .page -}} {{- $taxonomy := .taxonomy -}} {{- $isTags := eq $taxonomy "tags" -}} {{- if ne $page.Type "page" -}} {{- $terms := index $page.Params $taxonomy | default slice -}} {{- if $isTags -}} {{- $terms = complement ($page.Params.categories | default slice) $terms -}} {{- end -}} {{- range $i, $name := $terms -}} {{- if $isTags -}} #{{ $name }} {{- else -}} {{- if gt $i 0 -}}, {{- end -}} {{ $name }} {{- end -}} {{- end -}} {{- end -}}
