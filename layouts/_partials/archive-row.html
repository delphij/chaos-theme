{{- /* Renders one row of a dated timeline list: the archives page and the /series/ index. The list is grouped under year headings, so on narrow screens the date column drops the year and shows MM-DD. @context {page} page The page to list. @context {bool} seriesTitle Label the row with `series_title` when set, the full title kept as the tooltip (default: false). @context {bool} category Show the page's first category (default: false). @context {bool} series Show the page's first series when category is not shown (default: false). @context {bool} showDate Show the date column (default: true). @context {bool} fullDate Keep YYYY-MM-DD on narrow screens when rows are not grouped under year headings (default: false). @example: {{ partial "archive-row.html" (dict "page" . "category" true) }} */ -}} {{- $p := .page -}} {{- $showDate := true -}} {{- if isset . "showDate" -}} {{- $showDate = .showDate -}} {{- end -}} {{- $fullDate := .fullDate | default false -}} {{- $label := $p.Title -}} {{- $tooltip := "" -}} {{- if .seriesTitle -}} {{- with $p.Params.series_title -}} {{- $label = . -}} {{- if ne . $p.Title -}}{{- $tooltip = $p.Title -}}{{- end -}} {{- end -}} {{- end -}}
{{- if $showDate }} {{- if $fullDate }} {{ $p.Date.Format "2006-01-02" }} {{- else }} {{ $p.Date.Format "2006-01-02" }} {{ $p.Date.Format "01-02" }} {{- end }} {{- end }} {{ $label }} {{- if .category -}} {{- range first 1 ($p.GetTerms "categories") }} {{ .LinkTitle }} {{- end -}} {{- else if .series -}} {{- range first 1 ($p.GetTerms "series") }} {{ .LinkTitle }} {{- end -}} {{- end }}
