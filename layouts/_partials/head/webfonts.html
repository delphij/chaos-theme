{{- /* Opt-in self-hosted CJK webfonts (enabled via `[params.webfonts] enable = true`). Reads the pinned descriptor in `assets/data/webfonts.json` (shared with `tools/fetch_webfonts.py`), fingerprints each `assets/_fonts///font.css` via Hugo Pipes (publishing `font..css` alongside the `.woff2` slices with SRI `integrity`), and fills the `--webfont-*` slots in `tokens.css` / `prose.css` so font stacks stay DRY and locale `:lang()` rules stay intact. */ -}} {{- if site.Params.webfonts.enable -}} {{- with resources.Get "data/webfonts.json" -}} {{- with . | transform.Unmarshal -}} {{- $assetPrefix := strings.Trim (or .urlPrefix "_fonts") "/" -}} {{- $decls := slice -}} {{- range .fonts -}} {{- if eq .role "kai" -}} {{- $decls = $decls | append (printf `--webfont-kai:%q` .family) (printf `--webfont-kai-em:"%s Em"` .family) -}} {{- else if eq .role "serif" -}} {{- $decls = $decls | append (printf `--webfont-serif:%q` .family) (printf `--webfont-serif-em:"%s Em"` .family) -}} {{- end -}} {{- with resources.Get (printf "%s/%s/%s/font.css" $assetPrefix .slug .version) -}} {{- with . | fingerprint -}} {{- end -}} {{- end -}} {{- end -}} {{- if $decls }} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
