{{- /* Opt-in self-hosted CJK webfonts (enabled via `[params.webfonts] enable = true`). Reads the pinned descriptor in `assets/data/webfonts.json` (shared with `tools/fetch_webfonts.py`), loads each `assets/_fonts///font.css` non-blocking via Hugo Pipes `fingerprint` + SRI `integrity` ONLY on pages that actually use its role (`serif` -> `.vertical`; `kai` -> ``/``/`.mizige`), and fills the `--webfont-*` slots in `tokens.css` / `prose.css` so font stacks stay DRY and locale `:lang()` rules stay intact. */ -}} {{- if site.Params.webfonts.enable -}} {{- $page := . -}} {{- with resources.Get "data/webfonts.json" -}} {{- with . | transform.Unmarshal -}} {{- $assetPrefix := strings.Trim (or .urlPrefix "_fonts") "/" -}} {{- $needSerif := partial "_funcs/has-feature.html" (dict "page" $page "markup" `class="vertical`) -}} {{- $needKai := or (partial "_funcs/has-feature.html" (dict "page" $page "markup" "")) (partial "_funcs/has-feature.html" (dict "page" $page "markup" `class="mizige"`)) -}} {{- $decls := slice -}} {{- range .fonts -}} {{- $needed := true -}} {{- if eq .role "kai" -}} {{- $needed = $needKai -}} {{- if $needed -}} {{- $decls = $decls | append (printf `--webfont-kai:%q` .family) (printf `--webfont-kai-em:"%s Em"` .family) -}} {{- end -}} {{- else if eq .role "serif" -}} {{- $needed = $needSerif -}} {{- if $needed -}} {{- $decls = $decls | append (printf `--webfont-serif:%q` .family) (printf `--webfont-serif-em:"%s Em"` .family) -}} {{- end -}} {{- end -}} {{- if $needed -}} {{- with resources.Get (printf "%s/%s/%s/font.css" $assetPrefix .slug .version) -}} {{- with . | fingerprint -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $decls }} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
