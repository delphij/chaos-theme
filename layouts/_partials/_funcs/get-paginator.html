{{- /* Initializes or returns the paginator for paginated list pages. Called once from baseof.html, which stores the result as the page's "paginator" store entry; templates read that rather than .Paginator, so the custom collection (the filtered listing) is the one Hugo registers. Sections listed in site.Params.unpaginatedSections (default: the archives) are skipped: their layout renders a complete listing of its own, and paginating them would emit one identical full-size page per pager entry. @context {page} page The page context. @returns {paginator} The initialized paginator object, or false for non-paginated pages. */}} {{- $unpaginatedSections := site.Params.unpaginatedSections | default (slice "archives") -}} {{- $paginate := false -}} {{- if or .IsHome (eq .Kind "term") -}} {{- $paginate = true -}} {{- else if eq .Kind "section" -}} {{- $paginate = not (in $unpaginatedSections .Section) -}} {{- end -}} {{- $paginator := false -}} {{- if $paginate -}} {{- $paginator = .Paginate (partial "_funcs/get-list-pages.html" .) -}} {{- end -}} {{- return $paginator -}}
