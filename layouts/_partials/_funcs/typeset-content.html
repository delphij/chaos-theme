        {{- $sentEnd := `[\s,;:!?)]|”|<|$` -}}
        {{- $prose = $prose | replaceRE (printf `((?:^|[\s(>])[a-z]{2,}|)\.([A-Z][a-z]*(?:'|’|')?[a-z]*(?:%s))` $sentEnd) "$1. $2" -}}
        {{- $prose = $prose | replaceRE (printf `(^|[\s(>])([A-Z][a-z]{2,})\.((?:I(?:(?:'|’|')(?:[dms]|ve|ll|re))?|The|This|That|It(?:s|(?:'|’|')s)?|But|And|So|For|However)(?:%s))` $sentEnd) "$1$2. $3" -}}
      {{- end -}}
      {{- if and $holdCode $code -}}
        {{- $codeOut = $codeOut | append (printf "\x00\x01%d\x00" (len $codes)) $prose -}}
        {{- $codes = $codes | append $code -}}
      {{- else -}}
        {{- $codeOut = $codeOut | append $code $prose -}}
      {{- end -}}
    {{- end -}}
    {{- $seg = delimit $codeOut "" -}}
    {{- if $hasSwatch -}}
      {{- /* Swatch labels holding inline code ([`#hex`]{#hex}), with the
           code held in placeholders as for [text]{.} below. */ -}}
      {{- $seg = $seg | replaceRE $swatchCodeLabelRE $swatchLabelOut -}}
    {{- end -}}
    {{- if $hasEmDot -}}
      {{- /* Expand [text]{.} with  held in placeholders (counted as 1
           unit) so code inside emphasis dots works without matching literal
           [x]{.} inside , and allow balanced inline tags,  and
           color swatches without counting tag attributes against the 32-unit
           cap. */ -}}
      {{- $emUnit := printf `(?:[^\[\]<\x00\n]|\x00\x01[0-9]+\x00|<(?:a|strong|em|del|mark)\b[^>]*>[^\[\]<\n]*|]*>%s(?:[^\[\]<\n]|\x00\x01[0-9]+\x00|]*>)*|%s|[^\[\]<\n]*[^<\n]*[^<\n]*[^<\n]*)` $chipTag $chipTag -}}
      {{- $seg = $seg | replaceRE (printf `\[(%s{1,32})\]\{\.\}` $emUnit) `$1` -}}
    {{- end -}}
    {{- if $holdCode -}}
      {{- range $k, $c := $codes -}}
        {{- $seg = replace $seg (printf "\x00\x01%d\x00" $k) $c -}}
      {{- end -}}
    {{- end -}}
    {{- /* Spaces and line-wrap newlines at an inline tag boundary. The rules
         above match a CJK character and a letter separated only by whitespace;
         a tag between them ("见 ", "\n的") hides the pair, and
         Goldmark's eastAsianLineBreaks leaves newlines adjacent to ASCII
         Markdown delimiters ("[", ")", "`"), which HTML would collapse to a
         word space.

         Prose-styled tags (strong, em, del, mark, a, ruby) hold ordinary
         prose, which autospace treats like any other text: the space goes
         whatever the tag holds.

         Monospace tags (code, kbd, samp) lose the space next to an ideograph
         only when the character just inside the tag is a letter or digit.
         autospace puts no gap between an ideograph and a symbol, so for
         "使用 --bare" or "编辑 /usr/local" the typed
         space is the only separation, and inline code has no background to mark
         its edge instead. An escaped entity (<, ") counts as a symbol:
         its first character is "&". The closing side tests the last character
         before , so "a.out 的" loses its space and
         "~/ 下" keeps it. Next to full-width CJK punctuation
         ("， /404.html", ". ，"), the glyph already
         carries its own half-em blank space, so a typed space is always
         dropped. */ -}}
    {{- if $strip -}}
      {{- $seg = $seg | replaceRE (printf `(%s)\s+(<(?:strong|em|del|mark|a|ruby)\b)` $cjk) "$1$2" -}}
      {{- $seg = $seg | replaceRE (printf `()\s+(%s)` $cjk) "$1$2" -}}
      {{- $seg = $seg | replaceRE (printf `(%s)\s+(]*>%s)` $chipTag $cjk $cjk) "$1$2" -}}
      {{- $seg = $seg | replaceRE (printf `(%s)\s+(<(?:code|kbd|samp)\b)` $cjkPunct) "$1$2" -}}
      {{- $seg = $seg | replaceRE (printf `()\s+(%s)` $cjkPunct) "$1$2" -}}
      {{- $seg = $seg | replaceRE (printf `(%s)\s+(<(?:code|kbd|samp)\b[^>]*>[A-Za-z0-9])` $cjk) "$1$2" -}}
      {{- $seg = $seg | replaceRE (printf `([A-Za-z0-9])\s+(%s)` $cjk) "$1$2" -}}
    {{- end -}}
    {{- /* Work around the same Goldmark 1.8 css3draft rule 3 bug when a wrapped
         line opens with ASCII punctuation (“, ‘, **, *, [, `),
         dropping the soft break before an opening curly quote or inline AST
         node ("to“buy", "looking for...", " ->TIME_INS",
         "use--flag"). Trade-off: without the original newline,
         intentional unspaced intraword emphasis (un*believ*able), unspaced
         ASCII comma lists (`a`,`b`), or `foo(args)` would also
         receive a space; plain `(` stays untouched to preserve item(s),
         func(arg), and O(n). Remove once Goldmark is fixed. */ -}}
    {{- $inlineTags := `strong|em|del|mark|a|code|kbd|samp` -}}
    {{- $seg = $seg | replaceRE (printf `([A-Za-z0-9]|(?:[A-Za-z0-9)]|)[.,:]|\s->)(&(?:ldquo|lsquo);|(?:<(?:%s)\b[^>]*>)+(?:[A-Za-z0-9(/.$~-]|&(?:ldquo|lsquo|quot|#39);))` $inlineTags $inlineTags) "$1 $2" -}}
    {{- /* The same bug drops the break before a swatch's "[". Restore it only
         when Latin text sits on both sides, so autospace still handles a CJK
         label, and never after a hex code ("#9E3D3F[]{#9E3D3F}"), which an
         author places against its chip on purpose. */ -}}
    {{- $latinLead := `(?:<(?:strong|em|del|mark|a|code|kbd|samp)\b[^>]*>)*[A-Za-z0-9#(]` -}}
    {{- $seg = $seg | replaceRE (printf `(^|[^0-9A-Za-z#])([A-Za-z0-9]*[A-Za-z0-9,])(]*>%s%s|%s%s)` $chipTag $latinLead $chipTag $latinLead) "$1$2 $3" -}}
    {{- range (findRE `(?:[^<\n]{16,}|]*>https?://[^<\n]{24,})` $seg | uniq) -}}
      {{- $inner := replaceRE `^<[^>]+>|]+>$` "" . -}}
      {{- /* Match query &key= before _ or . inserts  inside the key name;
         skipping bare &ident preserves C &var and escaped &quot; literals */ -}}
      {{- $wrapped := $inner | replaceRE `&([A-Za-z0-9_~%-]+)=` "&${1}=" -}}
      {{- range seq 2 -}}
        {{- $wrapped = $wrapped | replaceRE `([A-Za-z0-9_)\]/;}])\.([A-Za-z_])` "${1}.${2}" -}}
      {{- end -}}
      {{- $wrapped = $wrapped | replaceRE `(/|\?|->)([A-Za-z0-9_~%])` "${1}${2}" -}}
      {{- /* Scope :: breaks only between non-colon-bounded identifiers, never
         IPv6. Each match consumes both adjacent :: delimiters, so one pass
         breaks every third scope; three passes complete a chain of any length */ -}}
      {{- range seq 3 -}}
        {{- $wrapped = $wrapped | replaceRE `(::|^|[^:A-Za-z0-9_])([A-Za-z_][A-Za-z0-9_]*)::([A-Za-z_~][A-Za-z0-9_]*)(::|[^:A-Za-z0-9_]|$)` "${1}${2}::${3}${4}" -}}
      {{- end -}}
      {{- range seq 2 -}}
        {{- $wrapped = $wrapped | replaceRE `([A-Za-z0-9_])\\([A-Za-z0-9_]{2,})` "${1}\\${2}" -}}
      {{- end -}}
      {{- /* Keep \/, ./, ../, ~/ and /. intact rather than splitting at the slash */ -}}
      {{- $wrapped = $wrapped | replaceRE `\\/` `\/` -}}
      {{- $wrapped = $wrapped | replaceRE `(^|[^/])/\.` "${1}/." -}}
      {{- $wrapped = $wrapped | replaceRE `(^|[^A-Za-z0-9_.])(\.{1,2})/` "${1}${2}/" -}}
      {{- $wrapped = $wrapped | replaceRE `(^|[^A-Za-z0-9_])~/` "${1}~/" -}}
      {{- range seq 2 -}}
        {{- $wrapped = $wrapped | replaceRE `\.\./\.\.` "../.." -}}
      {{- end -}}
      {{- /* Needs the preceding character, so adjacent matches overlap: a second pass catches a_b_c */ -}}
      {{- range seq 2 -}}
        {{- $wrapped = $wrapped | replaceRE `([A-Za-z0-9])_([A-Za-z0-9])` "${1}_${2}" -}}
      {{- end -}}
      {{- $wrapped = $wrapped | replaceRE `([A-Za-z0-9/+-])#([A-Za-z0-9_])` "${1}#${2}" -}}
      {{- if hasPrefix . "${1}" -}}
      {{- end -}}
      {{- /* Scheme "://" breaks after, not between its slashes; non-scheme "//"
         stays intact. No break where one already exists (token start, after a
         space) or right after an escaped quote/angle */ -}}
      {{- $wrapped = $wrapped | replaceRE `://` "://" -}}
      {{- $wrapped = $wrapped | replaceRE `//` "//" -}}
      {{- $wrapped = $wrapped | replaceRE `` "" -}}
      {{- $wrapped = $wrapped | replaceRE `(^|\s|&(?:lt|gt|quot|apos|#(?:[0-9]+|[xX][0-9a-fA-F]+));)` "${1}" -}}
      {{- if ne $wrapped $inner -}}
        {{- $seg = replace $seg . (replace . (printf ">%s<" $inner) (printf ">%s<" $wrapped)) -}}
      {{- end -}}
    {{- end -}}
    {{- /* Bound Blink's cross-tag text-autospace overflow fallback to the final
         CJK break unit while preserving strict kinsoku and adjacent full-width
         punctuation kerning (chws / text-spacing-trim). */ -}}
    {{- $cjkBreakable := `[\x{3042}\x{3044}\x{3046}\x{3048}\x{304a}-\x{3062}\x{3064}-\x{3082}\x{3084}\x{3086}\x{3088}-\x{308d}\x{308f}-\x{3094}\x{30a2}\x{30a4}\x{30a6}\x{30a8}\x{30aa}-\x{30c2}\x{30c4}-\x{30e2}\x{30e4}\x{30e6}\x{30e8}-\x{30ed}\x{30ef}-\x{30f4}\x{30f7}-\x{30fa}\x{3400}-\x{4dbf}\x{4e00}-\x{9fff}\x{f900}-\x{faff}\x{20000}-\x{323af}]` -}}
    {{- $cjkNoStart := `[\x{3005}\x{3041}\x{3043}\x{3045}\x{3047}\x{3049}\x{3063}\x{3083}\x{3085}\x{3087}\x{308e}\x{3095}\x{3096}\x{309d}\x{309e}\x{30a1}\x{30a3}\x{30a5}\x{30a7}\x{30a9}\x{30c3}\x{30e3}\x{30e5}\x{30e7}\x{30ee}\x{30f5}\x{30f6}\x{30fc}-\x{30fe}]` -}}
    {{- $cjkUnit := printf `%s%s*` $cjkBreakable $cjkNoStart -}}
    {{- $cjkChar := `[\x{3005}\x{3041}-\x{3096}\x{309d}\x{309e}\x{30a1}-\x{30fe}\x{3400}-\x{4dbf}\x{4e00}-\x{9fff}\x{f900}-\x{faff}\x{20000}-\x{323af}]` -}}
    {{- $cjkEndPunct := `[\x{3001}\x{3002}\x{ff0c}\x{ff0e}\x{ff1b}\x{ff1a}\x{ff01}\x{ff1f}\x{ff09}\x{3009}\x{300b}\x{300d}\x{300f}\x{3011}\x{3015}\x{3017}\x{3019}\x{301b}\x{301e}\x{301f}\x{ff3d}\x{ff5d}\x{2019}\x{201d}]` -}}
    {{- $cjkInsep := `[\x{2014}\x{2026}]` -}}
    {{- $openPunct := `[\x{3008}\x{300a}\x{300c}\x{300e}\x{3010}\x{3014}\x{3016}\x{3018}\x{301a}\x{301d}\x{ff08}\x{ff3b}\x{ff5b}\x{2018}\x{201c}]` -}}
    {{- $inlineStart := `(?:<(?:a|code|kbd|samp|strong|em|del|mark|ruby)\b[^>\n]*>)+[A-Za-z0-9]` -}}
    {{- $seg = $seg | replaceRE (printf `(%s|%s|%s)(%s(?:%s+%s+%s)?%s)` $cjkChar $cjkEndPunct $cjkInsep $cjkUnit $cjkEndPunct $openPunct $cjkUnit $inlineStart) "${1}${2}" -}}
    {{- $seg = $seg | replaceRE (printf `(%s|%s)(%s+%s%s)` $cjkChar $cjkInsep $openPunct $cjkUnit $inlineStart) "${1}${2}" -}}
    {{- if in $seg "color-swatch-chip" -}}
      {{- /* Bind every swatch chip to its opening unit (or bare following
           token) with white-space: nowrap so Blink cannot strand a lonely
           chip at the end of a line while long labels can still wrap.

           Pass 1 binds the chip with a whole following unit, but only a short
           one: a ruby base, inline code or tag holding at most 15 characters,
           an entity, or one character. An uncapped unit would put a whole
           label under nowrap, so
             [**这是一段相当长的加粗色名标签……**]{#9E3D3F}
           could not wrap and overflowed narrow screens; long inline code
           (16+ characters, already holding ) was not matched at all and
           left []{#165E83}`/usr/local/share/very/long` unbound.

           Pass 2 handles what pass 1 left: a chip followed by a longer
           , , , , ... moves inside that element and binds
           to its first character only:
             [chip]这是一段……
             [chip]/usr/local…
           The chip holds no text, so copying the link or code is unchanged.
           A longer ruby is left unbound instead: a chip inside the ruby base
           would shift the  annotation off its characters.

           A bound chip's "(?:<(?:strong|em|del|mark|a|b|i|s|u)\b[^>]*>)*[^<\n]{1,15}(?:)*[^<\n]*[^<\n]*[^<\n]*` -}}
      {{- $leadUnit := printf `(?:%s|&[A-Za-z0-9#]+;|[^\s<])` $shortRuby -}}
      {{- $bindUnit := printf `(?:%s|<(?:%s)\b[^>]*>[^<\n]{1,15}|%s)` $shortRuby $inlineOpen $inlineOpen $leadUnit -}}
      {{- $chipParts := `(]*>)` -}}
      {{- $bound := "${2}\x00\x05${3}" -}}
      {{- $seg = $seg | replaceRE (printf `()%s(%s)` $chipParts $bindUnit) (printf "%s${4}" $bound) -}}
      {{- $seg = $seg | replaceRE (printf `()%s((?:<(?:%s)\b[^>]*>)+)(%s)` $chipParts $inlineOpen $leadUnit) (printf "${4}%s${5}" $bound) -}}
      {{- $seg = replace $seg "\x00\x05" "" -}}
    {{- end -}}
    {{- if $katexHeld -}}
      {{- $rParts := split $seg "\x00\x02" -}}
      {{- $rOut := slice (index $rParts 0) -}}
      {{- range after 1 $rParts -}}
        {{- $rp := split . "\x00" -}}
        {{- $rOut = $rOut | append (index $katexHeld (int (index $rp 0))) (delimit (after 1 $rp) "\x00") -}}
      {{- end -}}
      {{- $seg = delimit $rOut "" -}}
    {{- end -}}
  {{- end -}}
  {{- $out = $out | append $pre $seg -}}
{{- end -}}
{{- $content = delimit $out "" -}}
{{- /* Restore last-cut first: a {.keep-space} block cut after a {.mizige}
     paragraph it contains carries that paragraph's placeholder, which only
     appears once the outer block is back. */ -}}
{{- if $kept -}}
  {{- range $k := seq (sub (len $kept) 1) 0 -}}
    {{- $content = replace $content (printf "\x00\x03%d\x00" $k) (index $kept $k) -}}
  {{- end -}}
{{- end -}}
{{- /* Summaries truncated at  carry footnote references whose
     definitions live at the end of the post; drop them when no footnote list
     is present so list pages and descriptions do not emit dead anchors,
     duplicate fnref IDs, or stray numbers. */ -}}
{{- if and $stripOrphanFootnotes (not (in $content `class="footnotes"`)) -}}
  {{- $content = $content | replaceRE `]*class="footnote-ref"[^>]*>.*?` "" -}}
{{- end -}}

{{- return $content -}}
