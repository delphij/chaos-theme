{{- /* Returns a slice of absolute URLs for page images, with site-level fallbacks. Search order: 1. Front matter "images" param 2. Page bundle image resources matching "feature*", "cover*", "thumbnail*", or first image 3. Site params "defaultImage" or "images" 4. Theme default fallback image ("/images/default-card.png") Bundle images are served as a derivative capped at 1200px rather than as the source file, both because social cards do not need more and because it keeps the source unreferenced (see build.publishResources). @context {page} page The page context. @returns {slice} Slice of absolute URL strings. @example: {{ $images := partial "_funcs/get-page-images.html" . }} */}} {{- $images := slice -}} {{- with .Params.images -}} {{- range . -}} {{- $images = $images | append (. | absURL) -}} {{- end -}} {{- end -}} {{- if eq (len $images) 0 -}} {{- /* Skip SVG resources: Hugo cannot resize them and Open Graph / Twitter Cards only accept raster images. */ -}} {{- with .Resources.ByType "image" -}} {{- $candidate := false -}} {{- with .GetMatch "{*feature*,*cover*,*thumbnail*}" -}} {{- if ne .MediaType.SubType "svg" -}} {{- $candidate = . -}} {{- end -}} {{- end -}} {{- if not $candidate -}} {{- range . -}} {{- if and (not $candidate) (ne .MediaType.SubType "svg") -}} {{- $candidate = . -}} {{- end -}} {{- end -}} {{- end -}} {{- with $candidate -}} {{- $card := . -}} {{- if ne .MediaType.SubType "gif" -}} {{- $width := cond (gt .Width 1200) 1200 .Width -}} {{- $card = .Resize (printf "%dx webp" $width) -}} {{- end -}} {{- $images = $images | append $card.Permalink -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq (len $images) 0 -}} {{- with or site.Params.defaultImage (index site.Params.images 0) -}} {{- $images = $images | append (. | absURL) -}} {{- end -}} {{- end -}} {{- if eq (len $images) 0 -}} {{- $images = $images | append ("images/default-card.png" | absURL) -}} {{- end -}} {{- return $images -}}
