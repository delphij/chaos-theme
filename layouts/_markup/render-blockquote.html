{{- /* Renders a blockquote, including GitHub-style alerts. Whitespace control is deliberate throughout. A render hook's output is baked into .Content, which the feed templates carry verbatim inside a CDATA section; the XML minifier does not descend into character data, so every newline and every indent a pretty-printed hook emits survives into the feed. The HTML page hides this because --minify collapses it afterwards - the feed is where the hook's real output shows. So the markup is emitted without any whitespace the document does not need, and both output formats get the same clean result. An alert's title is whatever Hugo finds on the first line of its first paragraph. With markup.goldmark.extensions.cjk.eastAsianLineBreaks enabled, every soft line break in that paragraph is dropped before Hugo looks, so the whole paragraph reads as one line and becomes the title: > [!TIP] > This is the body. -> title "This is the body.", no body So the hook reads the marker line back from the source: text after [!TYPE] there is the real title, and anything .AlertTitle holds beyond it is body text and is restored as the first paragraph. Where the source cannot be read (a file outside the project, such as a module mount), the title stands as Hugo gave it. @context {blockquote} The blockquote render hook context. */ -}} {{- $emojis := dict "caution" ":warning:" "disclaimer" ":warning:" "historical" ":scroll:" "important" ":exclamation:" "note" ":notebook:" "tip" ":bulb:" "warning" ":rotating_light:" -}} {{- if eq .Type "alert" -}} {{- $default := or (i18n .AlertType) (title .AlertType) -}} {{- $title := $default -}} {{- $lead := "" -}} {{- with .AlertTitle -}} {{- $title = . -}} {{- /* os.ReadFile takes paths relative to the project, not absolute ones */ -}} {{- $file := strings.TrimPrefix (printf "%s/" hugo.WorkingDir) $.Position.Filename -}} {{- if and $file (gt $.Position.LineNumber 0) (fileExists $file) -}} {{- $line := index (split (os.ReadFile $file) "\n") (sub $.Position.LineNumber 1) -}} {{- $own := trim (replaceRE `^\s*>\s*\[![A-Za-z]+\][+-]?` "" $line) " \t\r" -}} {{- if not $own -}} {{- $title = $default -}} {{- $lead = . -}} {{- else if and (hasPrefix . $own) (ne . $own) -}} {{- $title = $own -}} {{- $lead = trim (strings.TrimPrefix $own .) " " -}} {{- end -}} {{- end -}} {{- end -}}
