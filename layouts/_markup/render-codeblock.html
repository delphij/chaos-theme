{{- /* Renders a fenced code block through Chroma. Line numbers are on by default only from five lines up; a fence that sets linenos itself (which lands in .Options, not .Attributes) keeps its choice. Only that default is passed on: transform.HighlightCodeBlock merges the fence's own options already, and handing .Options back to it applies hl_lines' line-number offset a second time, highlighting the line below. @context {codeblock} The code block render hook context. */ -}} {{- $opts := dict -}} {{- if not (isset .Options "linenos") -}} {{- $opts = dict "linenos" (ge (len (split .Inner "\n")) 5) -}} {{- end -}} {{- /* Keyboard and screen-reader cleanup of Chroma's markup. Chroma puts tabindex="0" on every 
 it writes, the line-number
       column's included, so a numbered block cost two Tab stops before its
       copy button. The two 
s are not alike:

       - The line-number 
 never scrolls on its own (the code column
         does), and its numbers are noise to a screen reader. Its tabindex is
         dropped here, and unless the numbers are links (anchorlinenos) its
         cell is aria-hidden.
       - The code 
 is a horizontal scroller (overflow-x: auto in
         prose.css). WCAG 2.1.1 wants a scroller reachable from the keyboard,
         and Safari, unlike Chrome 130+ and Firefox, does not make one
         focusable by itself; that is why Chroma adds the tabindex. It stays
         in the HTML, so a page without JavaScript (and every feed reader)
         keeps it, and modules/code-copy.js (or xml-xsl-feed.js on the feed
         preview) removes it from a block whose lines all fit, where it would
         be a Tab stop that does nothing.

       The table is only layout (prose.css turns it into a flex row), so it is
       marked presentational instead of being read as a one-row table.

       Chroma emits the line-number cell as  followed by
       a newline and its 
; the \s* allows for that whitespace changing in
       a Chroma upgrade without the match silently failing. Fenced blocks
       highlighted outside this hook ({{< highlight >}}) get the same tabindex
       treatment from code-copy.js.

       Opt-in soft wrapping ({.wrap}) puts class="wrap" on .highlight via
       Chroma (falling back to the plain "text" lexer when the fence names no
       language, so bare fences still get per-line .line / .cl wrappers);
       sites or individual pages can also make wrapping the default via
       site.Params.codeWrap or page.Params.codeWrap, with {.nowrap} overriding
       it per fence. Since a wrapped block never scrolls horizontally, its code
       
 drops tabindex="0" up front too. transform.Highlight does not carry
       the fence's attributes, so the text-lexer path copies all of them onto
       .highlight itself, as HighlightCodeBlock would (code-vars.js reads
       code_vars="false" there).

       A wrapped block hides the line-number column and numbers its rows with
       a CSS counter, which would also hide anchorlinenos' links and their
       fragment targets; with anchorlinenos it takes Chroma's inline numbers
       instead, which stay in each row beside the wrapped text. */ -}}
{{- $classes := split (.Attributes.class | default "") " " -}}
{{- $wrap := site.Params.codeWrap | default false -}}
{{- if isset .Page.Params "codewrap" -}}
  {{- $wrap = .Page.Params.codewrap -}}
{{- end -}}
{{- if in $classes "nowrap" -}}
  {{- $wrap = false -}}
{{- else if in $classes "wrap" -}}
  {{- $wrap = true -}}
{{- end -}}
{{- $anchorLinenos := and .Options.anchorlinenos (ne (printf "%v" .Options.anchorlinenos) "false") -}}
{{- $linenos := cond (isset .Options "linenos") .Options.linenos $opts.linenos -}}
{{- if and $wrap $anchorLinenos $linenos (ne (printf "%v" $linenos) "false") -}}
  {{- $opts = merge $opts (dict "linenos" "inline") -}}
{{- end -}}
{{- $hl := "" -}}

  {{- $attrs := printf ` class="%s"` (htmlEscape (printf "highlight %s" $classList)) -}}
  {{- range $k, $v := .Attributes -}}
    {{- if ne $k "class" -}}
      {{- $attrs = printf `%s %s="%s"` $attrs $k (htmlEscape (printf "%v" $v)) -}}
    {{- end -}}
  {{- end -}}

         it they would be bare numbers shared by every block on the page. */ -}}
  {{- $anchors := dict "lineanchors" (printf "hl-%d" .Ordinal) -}}
  {{- $hl = string (transform.Highlight .Inner "text" (merge $anchors .Options $opts)) -}}
